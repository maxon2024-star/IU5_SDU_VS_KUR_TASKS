## 1. Введение и теоретическая база

На данный момент наш процессор умеет выполнять вычисления и работать с оперативной памятью (RAM). Однако настоящий микроконтроллер должен уметь взаимодействовать с внешним миром: зажигать светодиоды, читать состояния кнопок, общаться по шинам данных (UART, SPI).

Чтобы процессор мог управлять периферией, не создавая для этого специальных новых инструкций, используется архитектурный подход **Memory-Mapped I/O (MMIO — Ввод-вывод, отображаемый на память)**.
Суть MMIO гениально проста: мы выделяем определенные адреса в едином адресном пространстве, но физически эти адреса указывают не на ячейки RAM, а на регистры периферийных устройств (например, порта GPIO). Таким образом, обычная инструкция `STORE R1, [0xFF]` запишет данные не в память, а зажжет светодиоды!

**Цель работы:** Превратить процессорное ядро в полноценную Систему на Кристалле (SoC), добавив дешифратор адреса, порт GPIO и написав "прошивку" (Firmware) для работы с внешним миром.
**Теоретическая база:** Модуль 3 (Периферийные устройства, порты ввода-вывода).

### 3.1 Структурная схема SoC с MMIO

```text
                               +----------------+
                               |    Data RAM    |
                         +---->| (0x00 .. 0x7F) |----+
                         |     +----------------+    |
  +--------+   addr      |                           v    data_in
  |        |====/========+                        +-----+   /
  |  CPU   |  we         |     +----------------+ | MUX |==/==> В процессор
  | (Core) |---->+       +---->|   GPIO Port    | +-----+
  |        |     |       |     | (0x80 .. 0xFF) |----+
  +--------+     |       |     +----------------+    |
   |      ^      |       |      | LEDs    ^ Sw       |
   | data |      |     +---+    v         |          |
   +--/---+      +---->|Dec|  (Внешний мир)          |
   (out)               +---+                         |
                         | we_ram                    |
                         +---------------------------+
                         | we_gpio

```

### 3.2 Дешифратор адреса и мультиплексор чтения (`mmio_decoder.v`)

Этот модуль перехватывает запросы процессора к памяти.
Допустим, по вашему варианту MMIO Base = `0x80` (в двоичном виде `1000_0000`). Значит, если старший бит адреса равен `1`, мы обращаемся к периферии.

```verilog
module mmio_decoder #(
    parameter DATA_WIDTH = 16,
    parameter ADDR_WIDTH = 8
)(
    // Сигналы от процессора
    input  wire                  cpu_we,
    input  wire [ADDR_WIDTH-1:0] cpu_addr,
    
    // Сигналы к устройствам
    output wire                  we_ram,
    output wire                  we_gpio,
    
    // Данные от устройств (для маршрутизации обратно в процессор)
    input  wire [DATA_WIDTH-1:0] ram_data_out,
    input  wire [DATA_WIDTH-1:0] gpio_data_out,
    
    // Данные для процессора
    output reg  [DATA_WIDTH-1:0] cpu_data_in
);

    // Пример для MMIO Base = 0x80 (Старший бит = 1)
    wire is_mmio = (cpu_addr >= 8'h80);

    // 1. Дешифратор записи (Куда отправляем сигнал Write Enable?)
    assign we_ram  = cpu_we & ~is_mmio; // Пишем в RAM, если адрес < 0x80
    assign we_gpio = cpu_we &  is_mmio; // Пишем в GPIO, если адрес >= 0x80

    // 2. Мультиплексор чтения (Откуда отдаем данные процессору?)
    always @(*) begin
        if (is_mmio) begin
            cpu_data_in = gpio_data_out;
        end else begin
            cpu_data_in = ram_data_out;
        end
    end

endmodule

```

### 3.3 Модуль GPIO (`gpio.v`)

Это простейшая периферия. Запись по этому адресу меняет состояния выходных пинов (Светодиоды), а чтение возвращает состояния входных пинов (Кнопки/Переключатели).

```verilog
module gpio_port #(
    parameter DATA_WIDTH = 16
)(
    input  wire                  clk,
    input  wire                  rst,
    input  wire                  we,
    input  wire [DATA_WIDTH-1:0] data_in,   // Данные от процессора
    
    output wire [DATA_WIDTH-1:0] data_out,  // Данные для процессора
    
    // Физические пины микроконтроллера
    input  wire [7:0]            switches,  // Внешние переключатели
    output reg  [7:0]            leds       // Внешние светодиоды
);

    // Реакция на запись (Команда STORE)
    always @(posedge clk) begin
        if (rst) begin
            leds <= 8'b0;
        end else if (we) begin
            leds <= data_in[7:0]; // Зажигаем светодиоды младшим байтом
        end
    end

    // Реакция на чтение (Команда LOAD)
    // Процессор читает состояние переключателей
    assign data_out = {8'b0, switches};

endmodule

```

### 3.4 Пример прошивки микроконтроллера (`firmware.bin`)

Напишем программу, которая постоянно читает переключатели. Если переключатели в состоянии `0`, она зажигает светодиоды узором `0xAA`, иначе выводит на светодиоды то, что считала с переключателей.
*(Внимание: адреса и опкоды зависят от вашей ISA)*.

```text
// Адреса устройств:
// 0x80 - Порт GPIO

0001_0001_1000_0000 // 00: LDI R1, 0x80   (Загрузить в R1 адрес GPIO)
0001_0010_0000_0000 // 01: LDI R2, 0x00   (Загрузить в R2 константу 0 для сравнения)
0001_0011_1010_1010 // 02: LDI R3, 0xAA   (Загрузить в R3 узор 10101010)

// --- MAIN LOOP ---
1010_0100_0001_0000 // 03: LOAD R4, [R1]  (Прочитать состояние кнопок из GPIO в R4)
0100_0000_0100_0010 // 04: CMP R4, R2     (Сравнить кнопки с нулем. Если равны, Zero=1)
1000_0000_0000_0111 // 05: JZ 0x07        (Прыжок на адрес 07, если Zero=1)

// Если не ноль:
1100_0000_0001_0100 // 06: STORE R4, [R1] (Вывести на LEDs значение с кнопок)
1001_0000_0000_0011 // 07: JMP 0x03       (Вернуться в начало цикла)

// Если ноль (сюда прыгает JZ):
1100_0000_0001_0011 // 08: STORE R3, [R1] (Вывести на LEDs узор 0xAA)
1001_0000_0000_0011 // 09: JMP 0x03       (Вернуться в начало цикла)

```

### 3.5 Обновленный системный тестбенч (`soc_tb.v`)

В тестбенче мы теперь имитируем не только тактовый сигнал, но и действия пользователя.

```verilog
`timescale 1ns / 1ps

module soc_tb;
    reg clk;
    reg rst;
    reg [7:0] external_switches;
    wire [7:0] external_leds;

    // Ваш Top-Level теперь включает CPU, RAM, ROM, Дешифратор и GPIO
    microcontroller_top dut (
        .clk(clk),
        .rst(rst),
        .switches(external_switches),
        .leds(external_leds)
    );

    always #5 clk = ~clk;

    initial begin
        $dumpfile("soc.vcd");
        $dumpvars(0, soc_tb);

        clk = 0; rst = 1; external_switches = 8'h00;
        #15 rst = 0;
        
        $display("SoC Запущен. Кнопки: 0x00");
        #100; // Ждем 10 тактов (должен загореться узор 0xAA)
        
        // Пользователь переключил тумблеры!
        external_switches = 8'hF0; 
        $display("Пользователь изменил тумблеры на 0xF0");
        #100; // Процессор должен считать новые данные и вывести 0xF0 на светодиоды

        $finish;
    end

    // Отладка периферии
    always @(external_leds) begin
        $display("Time %0t: LEDs изменили состояние на -> %b", $time, external_leds);
    end
endmodule

```
